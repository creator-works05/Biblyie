         let mainTagButtonGrid= document.querySelector("#main-tag-button-grid");
        async function fetchTags() {
            try {
                let response = await fetch("http://localhost:3000/maintagCategory");
                let fetchedData = await response.json();
                fetchedData.forEach(element => {
                    let splitArray=element.main.split("#")

                    mainTagButtonGrid.innerHTML+=`<button>${splitArray[1]}</button>`

                });

            }
            catch (e) {
                 console.log("error occurred while fetching tags", e.message)
             }
       }
       fetchTags();
